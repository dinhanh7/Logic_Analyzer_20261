cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
  set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../third_party/pico-sdk")
endif()
if(NOT DEFINED PICO_BOARD)
  set(PICO_BOARD pico2_w)
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(rp2350_la C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(rp2350_la main.c tft_st7735.c decoders.c)
set_target_properties(rp2350_la PROPERTIES LINKER_LANGUAGE C)
target_include_directories(rp2350_la PRIVATE ../protocol)
pico_generate_pio_header(rp2350_la ${CMAKE_CURRENT_LIST_DIR}/logic_capture.pio)
target_link_libraries(rp2350_la pico_stdlib pico_stdio_usb hardware_pio hardware_dma hardware_clocks hardware_spi hardware_gpio)
pico_enable_stdio_usb(rp2350_la 1)
pico_enable_stdio_uart(rp2350_la 0)
target_compile_definitions(rp2350_la PRIVATE PICO_STDIO_USB_ENABLE_IRQ_BACKGROUND_TASK=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=0)
pico_add_extra_outputs(rp2350_la)
